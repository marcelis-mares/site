/* Monta ou refaz a aba Legenda. Pode ser chamada à mão pelo editor a qualquer
   momento: ela apaga e reescreve, então nunca duplica. */
function montarLegenda() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var a = ss.getSheetByName(ABA_LEGENDA);
  if (!a) a = ss.insertSheet(ABA_LEGENDA);
  a.clear();

  var linhas = [['codigo', 'grupo', 'animal', 'temperamento', 'numero', 'caracteristica', 'versao']];
  ORDEM_GRUPOS.forEach(function (g) {
    ITENS[g].forEach(function (texto, i) {
      linhas.push([g + (i < 10 ? '0' : '') + i, g, ANIMAIS[g], NOMES[g], i, texto, ITENS_VERSAO]);
    });
  });

  a.getRange(1, 1, linhas.length, linhas[0].length).setValues(linhas);
  a.getRange(1, 1, 1, linhas[0].length).setFontWeight('bold');
  a.setFrozenRows(1);
  a.autoResizeColumn(6);
  return linhas.length - 1;
}

var ORDEM_GRUPOS = ['S', 'C', 'M', 'F'];

/* Quantos respondentes uma turma precisa ter para o painel mostrar as palavras.
   Abaixo disso, um grupo pequeno viraria retrato de indivíduo. */
var MINIMO_PALAVRAS = 4;

/* Agrega as características marcadas de uma turma. Devolve apenas contagem por
   palavra, nunca quem marcou o quê. Empate é resolvido por uma regra declarada:
   primeiro a contagem, depois a ordem fixa dos grupos S, C, M, F, depois o
   número da característica. Nunca por ordem alfabética. */
function palavrasDaTurma(v, cab, alvo, iTime, iStatus) {
  var codigos = (alvo instanceof Array) ? alvo : limparLista(alvo);
  var idx = {};
  ORDEM_GRUPOS.forEach(function (g) { idx[g] = cab.indexOf('marcadas_' + g); });

  var contagem = {}, respondentes = 0;
  ORDEM_GRUPOS.forEach(function (g) {
    contagem[g] = [];
    for (var i = 0; i < ITENS[g].length; i++) contagem[g].push(0);
  });

  for (var r = 1; r < v.length; r++) {
    if (!pertence(v[r][iTime], codigos)) continue;
    if (iStatus >= 0 && v[r][iStatus] === 'blocked') continue;
    var contou = false;
    ORDEM_GRUPOS.forEach(function (g) {
      if (idx[g] < 0) return;
      var bruto = String(v[r][idx[g]] || '').trim();
      if (!bruto) return;
      contou = true;
      bruto.split(',').forEach(function (n) {
        var k = parseInt(n, 10);
        if (!isNaN(k) && k >= 0 && k < contagem[g].length) contagem[g][k]++;
      });
    });
    if (contou) respondentes++;
  }

  if (respondentes < MINIMO_PALAVRAS) {
    return { oculto: true, minimo: MINIMO_PALAVRAS, respondentes: respondentes };
  }

  var todas = [];
  ORDEM_GRUPOS.forEach(function (g, og) {
    contagem[g].forEach(function (n, i) {
      todas.push({ t: ITENS[g][i], g: g, n: n, og: og, i: i });
    });
  });

  var limpo = function (x) { return { t: x.t, g: x.g, n: x.n }; };

  /* Dentro de um mesmo empate de contagem, revezar entre os quatro grupos em vez
     de despejar todos de um grupo só. Sem isso, quando muita característica fica
     empatada em zero, a lista sai inteira de um temperamento e engana a leitura.
     A regra é declarada: primeiro a contagem, depois o rodízio S, C, M, F, depois
     o número da característica. Nunca ordem alfabética. */
  var selecionar = function (desc, quantas) {
    var niveis = {}, chaves = [];
    todas.forEach(function (x) {
      if (!niveis[x.n]) { niveis[x.n] = { S: [], C: [], M: [], F: [] }; chaves.push(x.n); }
      niveis[x.n][x.g].push(x);
    });
    chaves.sort(function (a, b) { return desc ? b - a : a - b; });

    var saida = [];
    for (var c = 0; c < chaves.length && saida.length < quantas; c++) {
      var balde = niveis[chaves[c]];
      ORDEM_GRUPOS.forEach(function (g) {
        balde[g].sort(function (a, b) { return a.i - b.i; });
      });
      var restam = true;
      for (var volta = 0; restam && saida.length < quantas; volta++) {
        restam = false;
        for (var g = 0; g < ORDEM_GRUPOS.length && saida.length < quantas; g++) {
          var fila = balde[ORDEM_GRUPOS[g]];
          if (volta < fila.length) { saida.push(limpo(fila[volta])); restam = true; }
        }
      }
    }
    return saida;
  };

  var zeradas = 0;
  todas.forEach(function (x) { if (x.n === 0) zeradas++; });

  var porPerfil = {};
  ORDEM_GRUPOS.forEach(function (g) {
    porPerfil[g] = todas.filter(function (x) { return x.g === g; })
      .sort(function (a, b) { return (b.n - a.n) || (a.i - b.i); })
      .slice(0, 5).map(limpo);
  });

  return {
    respondentes: respondentes,
    versao: ITENS_VERSAO,
    mais: selecionar(true, 15),
    menos: selecionar(false, 15),
    nunca_marcadas: zeradas,
    porPerfil: porPerfil
  };
}

/* Já recebeu o e-mail alguma vez? Evita mandar duas vezes para quem se inscreve
   de novo, o que é comum quando a pessoa não acha o link do grupo. */
function jaRecebeu(email) {
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_IMERSAO);
  if (!a || a.getLastRow() < 2) return false;
  var v = a.getRange(1, 1, a.getLastRow(), a.getLastColumn()).getValues();
  var cab = v[0], iEmail = cab.indexOf('email'), iEnv = cab.indexOf('email_enviado');
  if (iEmail < 0 || iEnv < 0) return false;
  var alvo = String(email || '').trim().toLowerCase();
  for (var i = 1; i < v.length; i++) {
    if (String(v[i][iEmail]).trim().toLowerCase() === alvo && String(v[i][iEnv]).trim()) return true;
  }
  return false;
}

function marcarEnviado(email) {
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_IMERSAO);
  if (!a || a.getLastRow() < 2) return;
  var cab = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var iEmail = cab.indexOf('email'), iEnv = cab.indexOf('email_enviado');
  if (iEmail < 0 || iEnv < 0) return;
  var col = a.getRange(2, iEmail + 1, a.getLastRow() - 1, 1).getValues();
  var alvo = String(email || '').trim().toLowerCase();
  var agora = new Date().toISOString();
  for (var i = 0; i < col.length; i++) {
    if (String(col[i][0]).trim().toLowerCase() === alvo) {
      a.getRange(i + 2, iEnv + 1).setValue(agora);
    }
  }
}

function enviarBoasVindas(d) {
  var email = String(d.email || '').trim();
  if (!email) return;
  if (jaRecebeu(email)) return;
  try {
    MailApp.sendEmail({
      to: email,
      subject: IMERSAO.assunto,
      htmlBody: emailImersao(d.nome),
      name: 'Marcelis Pereira · MARES',
      replyTo: COPIA_EQUIPE,
      bcc: COPIA_EQUIPE
    });
    marcarEnviado(email);
  } catch (err) {}
}

/* Roda à mão no editor. Manda o e-mail para todo mundo que está na aba Imersao
   e ainda não recebeu. Serve para alcançar quem se inscreveu antes disso existir:
   basta colar os inscritos na aba, com pelo menos nome e email preenchidos. */
function reenviarBoasVindas() {
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_IMERSAO);
  if (!a || a.getLastRow() < 2) return 'aba vazia';
  var v = a.getRange(1, 1, a.getLastRow(), a.getLastColumn()).getValues();
  var cab = v[0], iNome = cab.indexOf('nome'), iEmail = cab.indexOf('email'), iEnv = cab.indexOf('email_enviado');
  if (iEmail < 0 || iEnv < 0) return 'faltam colunas';

  var enviados = 0, jaFeitos = {};
  for (var i = 1; i < v.length; i++) {
    var email = String(v[i][iEmail] || '').trim();
    if (!email || String(v[i][iEnv]).trim()) continue;
    if (jaFeitos[email.toLowerCase()]) continue;
    try {
      MailApp.sendEmail({
        to: email,
        subject: IMERSAO.assunto,
        htmlBody: emailImersao(v[i][iNome]),
        name: 'Marcelis Pereira · MARES',
        replyTo: COPIA_EQUIPE,
        bcc: COPIA_EQUIPE
      });
      a.getRange(i + 1, iEnv + 1).setValue(new Date().toISOString());
      jaFeitos[email.toLowerCase()] = true;
      enviados++;
    } catch (err) {}
  }
  return 'enviados: ' + enviados;
}

/* ============================================================================
   BÚSSOLA DE IA MARES
   ----------------------------------------------------------------------------
   A Bússola é um levantamento curto, seis perguntas, sem nota e sem ranking.
   Ela manda dois pacotes para cá:

   1. registro: 'evento'  → cai na aba Eventos, sem nome e sem e-mail, e é o que
      alimenta o painel de turma.
   2. registro: 'bussola' → cai na aba Bussola, com o cadastro completo e as
      respostas abertas. É a aba que a Marcelis lê antes de um encontro.

   A página manda o pacote 2 duas vezes: uma quando a pessoa termina o teste, e
   outra se ela escrever a mensagem final antes de ir para o WhatsApp. Por isso
   a gravação é por sessão: a segunda chamada atualiza a linha da primeira, em
   vez de criar linha repetida.
============================================================================ */

var ABA_BUSSOLA = 'Bussola';

var COLUNAS_BUSSOLA = [
  'data','sessao','time','nome','email','whatsapp','empresa','faturamento',
  'ferramentas','areas_de_uso','frequencia','impacto','nivel_impacto',
  'o_que_quer_aprender','duvida_ou_receio','mensagem',
  'observacoes','whatsapp_url','origem','avisado'
];

function gravarBussola(d) {
  var a = aba(ABA_BUSSOLA, COLUNAS_BUSSOLA);
  var linha = {
    data:                new Date(),
    sessao:              d.sessao || '',
    time:                limpar(d.time),
    nome:                d.nome || '',
    email:               d.email || '',
    whatsapp:            d.whatsapp || '',
    empresa:             d.empresa || '',
    faturamento:         d.faturamento || '',
    ferramentas:         d.ferramentas || '',
    areas_de_uso:        d.areas_de_uso || '',
    frequencia:          d.frequencia || '',
    impacto:             d.impacto || '',
    nivel_impacto:       d.nivel_impacto === undefined ? '' : d.nivel_impacto,
    o_que_quer_aprender: d.o_que_quer_aprender || '',
    duvida_ou_receio:    d.duvida_ou_receio || '',
    mensagem:            d.mensagem || '',
    observacoes:         d.observacoes || '',
    whatsapp_url:        d.whatsapp_url || '',
    origem:              d.origem || ''
  };

  var existente = linhaDaSessao(a, d.sessao);
  if (existente) {
    var jaAvisou = valorDaColuna(a, existente, 'avisado');
    var mensagemNova = !!linha.mensagem && !valorDaColuna(a, existente, 'mensagem');
    atualizarEm(a, COLUNAS_BUSSOLA, linha, existente);
    if (mensagemNova) {
      avisarEquipeBussola(linha, true);
      marcarAvisado(a, existente);
    }
    return;
  }

  gravarEm(a, COLUNAS_BUSSOLA, linha);

  /* Numa palestra com turma, dezenas de pessoas terminam ao mesmo tempo e o
     e-mail individual só atrapalharia, além de gastar a cota do MailApp. Nesse
     caso os dados ficam na planilha e no painel. Fora de turma, ou quando a
     pessoa escreveu uma mensagem, a equipe é avisada na hora. */
  if (!linha.time || linha.mensagem) {
    avisarEquipeBussola(linha, !!linha.mensagem);
    marcarAvisado(a, a.getLastRow());
  }
}

/* --------------------------------------------------------------------------
   Ajudantes de linha, usados pela Bússola e disponíveis para o que vier depois.
-------------------------------------------------------------------------- */

function linhaDaSessao(a, sessao) {
  if (!sessao || a.getLastRow() < 2) return 0;
  var cab = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var col = cab.indexOf('sessao') + 1;
  if (col < 1) return 0;
  var v = a.getRange(2, col, a.getLastRow() - 1, 1).getValues();
  for (var i = v.length - 1; i >= 0; i--) {
    if (String(v[i][0]) === String(sessao)) return i + 2;
  }
  return 0;
}

function valorDaColuna(a, linha, nome) {
  var cab = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var col = cab.indexOf(nome) + 1;
  if (col < 1) return '';
  return a.getRange(linha, col).getValue();
}

function marcarAvisado(a, linha) {
  var cab = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var col = cab.indexOf('avisado') + 1;
  if (col > 0) a.getRange(linha, col).setValue(new Date());
}

/* Atualiza a linha inteira sem apagar o que já estava lá: campo que chega
   vazio conserva o valor antigo. */
function atualizarEm(a, colunas, d, linha) {
  var cabecalho = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var faltando = colunas.filter(function (c) { return cabecalho.indexOf(c) < 0; });
  if (faltando.length) {
    a.getRange(1, cabecalho.length + 1, 1, faltando.length)
     .setValues([faltando]).setFontWeight('bold');
    cabecalho = cabecalho.concat(faltando);
  }
  var atual = a.getRange(linha, 1, 1, cabecalho.length).getValues()[0];
  var novo = cabecalho.map(function (c, i) {
    var v = d[c];
    return (v === undefined || v === null || v === '') ? atual[i] : v;
  });
  a.getRange(linha, 1, 1, cabecalho.length).setValues([novo]);
}

function escapar(s) {
  return String(s === undefined || s === null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

function doPost(e) {
  var dados;
  try {
    dados = JSON.parse(e.postData.contents);
  } catch (err) {
    return ContentService.createTextOutput('json invalido');
  }

  /* Numa palestra dezenas de pessoas terminam quase ao mesmo tempo. Sem a
     fila, duas chamadas simultâneas podem criar a aba e escrever o cabeçalho
     as duas, e a planilha fica com linha repetida. */
  var trava = LockService.getScriptLock();
  try { trava.waitLock(25000); } catch (err) { /* segue mesmo assim */ }
  try {
    return processar(dados);
  } finally {
    try { trava.releaseLock(); } catch (err) {}
  }
}

function processar(dados) {

  // Registro enxuto de qualquer teste, só para os painéis.
  if (dados.registro === 'evento') {
    gravarEvento(dados);
    return ContentService.createTextOutput('ok');
  }

  // Bússola de IA. Cadastro completo na aba própria, e aviso para a equipe.
  if (dados.registro === 'bussola') {
    gravarBussola(dados);
    return ContentService.createTextOutput('ok');
  }

  // Inscrição na Imersão. Grava na aba própria e manda o e-mail de boas-vindas.
  if (dados.registro === 'imersao') {
    gravarImersao(dados);
    enviarBoasVindas(dados);
    return ContentService.createTextOutput('ok');
  }

  gravar(dados);
  gravarEvento(dados);

  // O envio automático só acontece quando o site marcou o registro como sent.
  // Qualquer status diferente disso fica na planilha esperando revisão humana,
  // como manda o manual técnico.
  if (dados.status === 'sent' && dados.email && dados.email_html) {
    enviar(dados);
  }

  return ContentService.createTextOutput('ok');
}

/**
 * Painéis. Três modos:
 *   ?chave=X&modo=turmas          lista tudo o que já foi usado
 *   ?chave=X&time=codigo          dados do Teste dos Temperamentos daquela turma
 *   ?chave=X&modo=evento&time=Y   dados de qualquer teste daquela turma
 * Aceita callback para JSONP, que é o plano B do painel.
 */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var saida;

  if (p.chave !== CHAVE_PAINEL) {
    saida = { erro: 'chave invalida' };
  } else if (p.modo === 'turmas') {
    saida = listaDeTurmas();
  } else if (p.geral === '1') {
    /* Respondentes gerais: quem chegou pelo site, fora de qualquer turma. */
    saida = (p.teste && p.teste !== 'Temperamentos')
      ? eventosDaTurma([], p.teste)
      : dadosDaTurma([]);
  } else if (p.modo === 'evento') {
    saida = eventosDaTurma(limparLista(p.time), p.teste || '');
  } else if (!p.time) {
    saida = { erro: 'informe o codigo da turma' };
  } else {
    saida = dadosDaTurma(limparLista(p.time));
  }

  var txt = JSON.stringify(saida);
  if (p.callback) {
    return ContentService.createTextOutput(p.callback + '(' + txt + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(txt)
    .setMimeType(ContentService.MimeType.JSON);
}

/* Mesma limpeza que o site aplica no código do time. */
function limpar(v) {
  return String(v || '').trim().toLowerCase().replace(/[^a-z0-9-]/g, '').slice(0, 40);
}

/* Aceita um código só ou vários separados por vírgula, para reunir turmas
   diferentes num painel único. Devolve sempre uma lista, mesmo com um item só.
   O sinal de mais também vale como separador, porque o navegador às vezes
   transforma vírgula nele quando o link é colado de um lugar para outro. */
function limparLista(v) {
  var brutos = String(v || '').split(/[,+]/);
  var saida = [], vistos = {};
  for (var i = 0; i < brutos.length; i++) {
    var c = limpar(brutos[i]);
    if (c && !vistos[c]) { vistos[c] = true; saida.push(c); }
  }
  return saida;
}

/* Confere se a linha pertence a alguma das turmas pedidas. Lista vazia
   significa "sem turma nenhuma", que é o recorte dos respondentes gerais. */
function pertence(valor, codigos) {
  var c = String(valor || '').toLowerCase().trim();
  if (!codigos.length) return c === '';
  return codigos.indexOf(c) >= 0;
}

function aba(nome, colunas) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var a = ss.getSheetByName(nome);
  if (!a) a = ss.insertSheet(nome);
  if (a.getLastRow() === 0) {
    a.appendRow(colunas);
    a.setFrozenRows(1);
    a.getRange(1, 1, 1, colunas.length).setFontWeight('bold');
  }
  return a;
}

/**
 * Grava sempre pelo nome da coluna, nunca pela posição. Se o site passar a
 * enviar um campo novo, ele é acrescentado ao final do cabeçalho e as linhas
 * antigas continuam alinhadas.
 */
function gravarEm(a, colunas, d) {
  var cabecalho = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var faltando = colunas.filter(function (c) { return cabecalho.indexOf(c) < 0; });
  if (faltando.length) {
    a.getRange(1, cabecalho.length + 1, 1, faltando.length)
     .setValues([faltando]).setFontWeight('bold');
    cabecalho = cabecalho.concat(faltando);
  }
  a.appendRow(cabecalho.map(function (c) { return d[c] === undefined ? '' : d[c]; }));
}

function gravar(d) {
  gravarEm(aba(ABA, COLUNAS), COLUNAS, d);
  /* A legenda nasce sozinha na primeira gravação e nunca mais é tocada, a não ser
     que alguém chame montarLegenda() de propósito para refazer. */
  if (!SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_LEGENDA)) {
    try { montarLegenda(); } catch (err) {}
  }
}

function gravarEvento(d) {
  var dim = d.dimensoes;
  /* A Bússola não tem nota nem pilar forte e fraco. O que ela tem de mais
     próximo é o nível de impacto que a pessoa enxerga, de 1 a 4, e a lista de
     ferramentas e de áreas. Sem isso a linha dela entraria vazia no painel. */
  if (!dim && (d.ferramentas || d.areas)) {
    dim = { ferramentas: d.ferramentas || [], areas: d.areas || [], frequencia: d.frequencia || '' };
  }
  var evento = {
    processed_at: d.processed_at || d.data || new Date().toISOString(),
    teste: d.teste || '',
    time: limpar(d.time),
    sessao: d.sessao || '',
    media: d.media || d.nivel_impacto || '',
    nivel: d.nivel || d.impacto || '',
    forte: d.dimensao_forte || d.pilar_forte || '',
    fraco: d.dimensao_fraca || d.pilar_fraco || '',
    predominante: d.predominante_animal || d.predominante || '',
    secundario: d.secundario_animal || d.secundario || '',
    classe: d.classe || '',
    escala: d.escala || '',
    dimensoes: dim ? (typeof dim === 'string' ? dim : JSON.stringify(dim)) : '',
    origem: d.origem || ''
  };
  gravarEm(aba(ABA_EVENTOS, COLUNAS_EVENTO), COLUNAS_EVENTO, evento);
}

/**
 * Lista tudo o que já foi usado, por teste e por código de turma. É isso que
 * permite descobrir uma turma que alguém da equipe criou sem avisar.
 */
function listaDeTurmas() {
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_EVENTOS);
  if (!a || a.getLastRow() < 2) return { turmas: [], gerado_em: new Date().toISOString() };

  var v = a.getRange(1, 1, a.getLastRow(), a.getLastColumn()).getValues();
  var cab = v[0];
  var iT = cab.indexOf('teste'), iC = cab.indexOf('time'), iD = cab.indexOf('processed_at');

  var mapa = {};
  for (var i = 1; i < v.length; i++) {
    if (String(v[i][iT]) === 'teste' && String(v[i][iC]) === 'time') continue; // cabeçalho repetido
    var teste = String(v[i][iT] || 'Sem teste');
    var codigo = String(v[i][iC] || '');
    var quando = String(v[i][iD] || '');
    var k = teste + '||' + codigo;
    if (!mapa[k]) mapa[k] = { teste: teste, time: codigo, respostas: 0, primeira: quando, ultima: quando };
    mapa[k].respostas++;
    if (quando && quando < mapa[k].primeira) mapa[k].primeira = quando;
    if (quando && quando > mapa[k].ultima) mapa[k].ultima = quando;
  }

  var lista = Object.keys(mapa).map(function (k) { return mapa[k]; });
  lista.sort(function (a, b) { return a.ultima < b.ultima ? 1 : -1; });
  return { turmas: lista, gerado_em: new Date().toISOString() };
}

function eventosDaTurma(codigo, teste) {
  var codigos = (codigo instanceof Array) ? codigo : limparLista(codigo);
  var rotulo = codigos.join(',');
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_EVENTOS);
  if (!a || a.getLastRow() < 2) return { time: rotulo, times: codigos, teste: teste, total: 0, linhas: [] };

  var v = a.getRange(1, 1, a.getLastRow(), a.getLastColumn()).getValues();
  var cab = v[0];
  var iC = cab.indexOf('time'), iT = cab.indexOf('teste');
  var expor = [];
  for (var c = 0; c < cab.length; c++) {
    if (cab[c] && NUNCA_EXPOR.indexOf(cab[c]) < 0) expor.push(c);
  }

  var linhas = [];
  for (var i = 1; i < v.length; i++) {
    if (String(v[i][iT]) === 'teste' && String(v[i][iC]) === 'time') continue; // cabeçalho repetido
    if (!pertence(v[i][iC], codigos)) continue;
    if (teste && String(v[i][iT]) !== teste) continue;
    var o = {};
    expor.forEach(function (c) { o[cab[c]] = v[i][c]; });
    linhas.push(o);
  }
  return { time: rotulo, times: codigos, teste: teste, total: linhas.length, gerado_em: new Date().toISOString(), linhas: linhas };
}

function dadosDaTurma(codigo) {
  var codigos = (codigo instanceof Array) ? codigo : limparLista(codigo);
  var rotulo = codigos.join(',');
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA);
  if (!a || a.getLastRow() < 2) return { time: rotulo, times: codigos, total: 0, linhas: [] };

  var v = a.getRange(1, 1, a.getLastRow(), a.getLastColumn()).getValues();
  var cab = v[0];
  var iTime = cab.indexOf('time');
  var iStatus = cab.indexOf('status');
  if (iTime < 0) return { time: rotulo, times: codigos, total: 0, linhas: [] };

  var expor = [];
  for (var c = 0; c < cab.length; c++) {
    if (cab[c] && NUNCA_EXPOR.indexOf(cab[c]) < 0) expor.push(c);
  }

  var linhas = [];
  for (var i = 1; i < v.length; i++) {
    if (!pertence(v[i][iTime], codigos)) continue;
    if (iStatus >= 0 && v[i][iStatus] === 'blocked') continue;
    var o = {};
    expor.forEach(function (c) { o[cab[c]] = v[i][c]; });
    linhas.push(o);
  }

  return {
    time: rotulo,
    times: codigos,
    total: linhas.length,
    gerado_em: new Date().toISOString(),
    linhas: linhas,
    palavras: palavrasDaTurma(v, cab, codigos, iTime, iStatus)
  };
}

  var opcoes = {
    htmlBody: d.email_html,

    replyTo: COPIA_EQUIPE
  };
  if (COPIA_EQUIPE) opcoes.bcc = COPIA_EQUIPE;

  try {
    MailApp.sendEmail(Object.assign({ to: d.email, subject: assunto }, opcoes));
  } catch (err) {
    marcarErro(d.sessao, String(err));
  }
}

function marcarErro(sessao, msg) {
  var a = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA);
  if (!a) return;
  var cab = a.getRange(1, 1, 1, a.getLastColumn()).getValues()[0];
  var col = cab.indexOf('review_reason') + 1;
  var colSessao = cab.indexOf('sessao') + 1;
  if (col < 1 || colSessao < 1) return;
  var v = a.getRange(1, colSessao, a.getLastRow(), 1).getValues();
  for (var i = v.length - 1; i >= 0; i--) {
    if (v[i][0] === sessao) {
      a.getRange(i + 1, col).setValue('falha no envio: ' + msg);
      return;
    }
  }
}
