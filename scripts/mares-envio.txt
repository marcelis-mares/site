/**
 * Envia usando o endereço definido em REMETENTE.
 * Se o script for do próprio atendimento@, sai direto.
 * Se for de outra conta que tenha o atendimento@ como alias verificado
 * em Gmail > Ver todas as configurações > Contas > Enviar e-mail como,
 * o alias é usado. Sem alias, cai no envio padrão da conta dona do script.
 */
function enviar(msg) {
  var eu = Session.getActiveUser().getEmail();
  if (REMETENTE && REMETENTE !== eu) {
    var aliases = [];
    try { aliases = GmailApp.getAliases(); } catch (e) {}
    if (aliases.indexOf(REMETENTE) !== -1) {
      GmailApp.sendEmail(msg.to, msg.subject, msg.body, {
        htmlBody: msg.htmlBody,
        name: msg.name,
        replyTo: msg.replyTo,
        from: REMETENTE
      });
      return;
    }
  }
  MailApp.sendEmail(msg);
}

/* ============ ENTRADA ============ */
function doPost(e) {
  var resposta = { ok: false };
  try {
    var d = JSON.parse(e.postData.contents);
    var aba = gravar(d);
    resposta.gravado = true;

    if (aba.novo) {
      try { enviar(montarEmailLead(d)); resposta.email = true; }
      catch (err) { resposta.erroEmail = String(err); }

      try { enviar(montarEmailInterno(d)); resposta.interno = true; }
      catch (err) { resposta.erroInterno = String(err); }
    } else {
      resposta.duplicado = true;
    }
    resposta.ok = true;
  } catch (err) {
    resposta.erro = String(err);
  }
  return ContentService.createTextOutput(JSON.stringify(resposta))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet() {

}

/* ============ PLANILHA ============ */
function gravar(d) {
  var nomeAba = (d.teste === 'Sonda') ? 'Sonda' : 'RaioX';
  var ss = SpreadsheetApp.openById(PLANILHA_ID);
  var aba = ss.getSheetByName(nomeAba) || ss.insertSheet(nomeAba);

  var colunas = ['data', 'sessao', 'nome', 'email', 'whatsapp', 'empresa',
                 'faturamento', 'funcionarios', 'momento', 'nivel', 'media',
                 'dimensao_forte', 'dimensao_fraca', 'mensagem', 'observacoes', 'origem'];

  if (aba.getLastRow() === 0) {
    aba.appendRow(colunas);
    aba.getRange(1, 1, 1, colunas.length).setFontWeight('bold');
    aba.setFrozenRows(1);
  }

  /* não grava nem manda e-mail duas vezes para a mesma sessão */
  if (d.sessao && aba.getLastRow() > 1) {
    var achou = aba.getRange(2, 2, aba.getLastRow() - 1, 1)
                   .createTextFinder(d.sessao).matchEntireCell(true).findNext();
    if (achou) return { novo: false };
  }

  aba.appendRow(colunas.map(function (c) {
    if (c === 'data') return new Date();
    return d[c] === undefined ? '' : d[c];
  }));
  return { novo: true };
}

/* barras de cada dimensão */
function blocoDimensoes(d) {
  var dims = d.dimensoes || [];
  if (!dims.length) return '';
  var linhas = dims.map(function (p) {
    var nota = parseFloat(String(p.nota).replace(',', '.')) || 0;
    var pct = Math.max(4, Math.round(nota / 5 * 100));
    var cor = p.faixa === 'baixo' ? '#c8502c' : (p.faixa === 'alto' ? '#15A5A8' : '#e0a33c');
    return ''
      + '<tr><td style="padding:0 0 14px 0;">'
      +   '<div style="font-size:14px;color:#22303c;padding-bottom:5px;">'
      +     '<strong>' + esc(p.nome) + '</strong> '
      +     '<span style="color:#7b8894;">' + esc(p.sub || '') + '</span>'
      +     '<span style="float:right;color:#5b6b78;">' + virgula(p.nota) + '</span>'
      +   '</div>'
      +   '<table role="presentation" width="100%" cellpadding="0" cellspacing="0" '
      +     'style="background:#e7ecf0;border-radius:4px;"><tr>'
      +     '<td width="' + pct + '%" style="background:' + cor + ';height:8px;border-radius:4px;font-size:0;line-height:0;">&nbsp;</td>'
      +     '<td style="font-size:0;line-height:0;">&nbsp;</td>'
      +   '</tr></table>'
      + '</td></tr>';
  }).join('');

  return ''
    + '<div style="padding:6px 32px 10px 32px;">'
    + '<div style="font-size:13px;letter-spacing:1px;text-transform:uppercase;color:#7b8894;padding-bottom:14px;">'
    +   'Suas dimensões</div>'
    + '<table role="presentation" width="100%" cellpadding="0" cellspacing="0">' + linhas + '</table>'
    + '</div>';
}

/* lista com as respostas dadas */
function blocoRespostas(d) {
  var lista = d.perguntas_respostas || [];
  if (!lista.length) return '';
  var itens = lista.map(function (q, i) {
    return '<tr>'
      + '<td style="padding:0 0 10px 0;font-size:13px;line-height:1.5;color:#5b6b78;">'
      +   '<strong style="color:#22303c;">' + (i + 1) + '. ' + esc(q.pergunta) + '</strong><br>'
      +   esc(q.resposta) + ' <span style="color:#9aa6b1;">(' + (q.nota || 0) + ' de 5)</span>'
      + '</td></tr>';
  }).join('');

  return ''
    + '<div style="padding:8px 32px 28px 32px;border-top:1px solid #e7ecf0;">'
    + '<div style="font-size:13px;letter-spacing:1px;text-transform:uppercase;color:#7b8894;padding:18px 0 14px 0;">'
    +   'O que você respondeu</div>'
    + '<table role="presentation" width="100%" cellpadding="0" cellspacing="0">' + itens + '</table>'
    + '</div>';
}

function textoSimples(d, primeiro, nomeTeste) {
  var l = [];
  l.push('Oi' + (primeiro ? ', ' + primeiro : '') + '.');
  l.push('');
  l.push('Seu ' + nomeTeste + ': ' + (d.nivel || '') + ', nota ' + virgula(d.media) + ' de 5,00.');
  l.push('');
  l.push(d.nivel_texto || '');
  l.push('');
  l.push('Dimensão mais forte: ' + (d.dimensao_forte || ''));
  l.push('Dimensão mais fraca: ' + (d.dimensao_fraca || ''));
  l.push('');
  l.push('Conversar comigo: ' + WHATSAPP);
  l.push('');
  l.push('Marcelis Pereira · MARES · Pessoas, Dados e Gestão');
  return l.join('\n');
}

/* ============ AVISO INTERNO ============ */
function montarEmailInterno(d) {
  var linhas = [
    ['Teste', d.teste || 'RaioX'],
    ['Nome', d.nome],
    ['Empresa', d.empresa],
    ['E-mail', d.email],
    ['WhatsApp', d.whatsapp],
    ['Faturamento', d.faturamento],
    ['Funcionários', d.funcionarios],
    ['Momento da empresa', d.momento],
    ['Nível', d.nivel],
    ['Nota', virgula(d.media)],
    ['Mais forte', d.dimensao_forte],
    ['Mais fraca', d.dimensao_fraca],
    ['Mensagem que escreveu', d.mensagem]
  ].filter(function (x) { return x[1]; })
   .map(function (x) {
     return '<tr><td style="padding:4px 12px 4px 0;color:#5b6b78;">' + x[0] + '</td>'
          + '<td style="padding:4px 0;"><strong>' + esc(String(x[1])) + '</strong></td></tr>';
   }).join('');

  var html = '<div style="font-family:Arial,Helvetica,sans-serif;font-size:14px;color:#22303c;">'
    + '<h2 style="margin:0 0 14px 0;">Novo resultado de ' + esc(d.teste || 'RaioX') + '</h2>'
    + '<table cellpadding="0" cellspacing="0">' + linhas + '</table>'
    + '<p style="margin:18px 0 6px 0;"><strong>Respostas completas</strong></p>'
    + '<p style="font-size:13px;line-height:1.6;color:#5b6b78;">' + esc(d.observacoes || '') + '</p>'
    + '<p><a href="' + WHATSAPP + '">Abrir conversa no WhatsApp</a></p>'
    + '</div>';

  return {
    to: AVISO_INTERNO,
    subject: 'Novo ' + (d.teste || 'RaioX') + ' · ' + (d.nome || '') + ' · ' + (d.empresa || ''),
    htmlBody: html,
    name: NOME_REMETENTE,
    replyTo: d.email || RESPONDER_PARA
  };
}

/* ============ APOIO ============ */
function esc(t) {
  return String(t === undefined || t === null ? '' : t)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

function virgula(n) {
  var v = parseFloat(String(n).replace(',', '.'));
  return isNaN(v) ? '' : v.toFixed(2).replace('.', ',');
}
